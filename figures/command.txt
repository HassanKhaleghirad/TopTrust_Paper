گام ۱: ذخیره‌ی لاگ اجرا

پس از اجرای مدل، لاگ را ذخیره کنید:
bash

python Edited_toptrust_version4.py \
  --intas --mode uniform --epsilon 8 --rounds 150 \
  --lr 0.15 --cmax 1.0 --intas_features edge_rel_kin \
  --seed 42 --report_test --report_unseen_test \
  2>&1 | tee train_log.txt

  --out figs/ \
